```typescript
// lib/validators.ts
/**
 * GPS coordinate validation with industry-standard precision
 * 
 * Precision standard: 5-6 decimal places
 * - 5 decimals = 1.1m accuracy (matches commercial GPS receivers)
 * - 6 decimals = 0.11m accuracy
 * - 7+ decimals exceed GPS hardware capability (<10mm) and waste storage
 * 
 * Source: coordinate-converter.com industry standard
 */

export interface GPSCoordinate {
  latitude: number
  longitude: number
}

/**
 * Validates and normalizes GPS coordinates with 6 decimal precision
 * @param lat - Latitude (-90 to 90)
 * @param lng - Longitude (-180 to 180)
 * @returns Normalized coordinate or null if invalid
 */
export function validateGPSCoordinate(
  lat: number,
  lng: number
): GPSCoordinate | null {
  // Validate ranges
  if (lat < -90 || lat > 90) return null
  if (lng < -180 || lng > 180) return null
  
  // Round to 6 decimal places (0.11m precision)
  const latitude = Math.round(lat * 1000000) / 1000000
  const longitude = Math.round(lng * 1000000) / 1000000
  
  return { latitude, longitude }
}

/**
 * Regex for validating GPS coordinate strings
 * Allows: -123.456789, 0, 90, -180.123456
 * Rejects: abc, 123.123.123, 91.0 (range validation happens in validateGPSCoordinate)
 */
export const GPS_REGEX = /^-?\d{1,3}(\.\d{1,6})?$/

/**
 * Parses GPS coordinate strings and validates
 * @param latStr - Latitude string
 * @param lngStr - Longitude string
 * @returns Validated coordinate or null if invalid format/range
 */
export function parseGPSString(
  latStr: string,
  lngStr: string
): GPSCoordinate | null {
  if (!GPS_REGEX.test(latStr) || !GPS_REGEX.test(lngStr)) {
    return null
  }
  
  const lat = parseFloat(latStr)
  const lng = parseFloat(lngStr)
  
  return validateGPSCoordinate(lat, lng)
}
```

import { describe, it, expect } from 'vitest'
import { validateGPSCoordinate, parseGPSString, GPS_REGEX } from '@/lib/validators'

describe('validateGPSCoordinate', () => {
  it('validates and normalizes valid coordinates', () => {
    const result = validateGPSCoordinate(40.7128, -74.0060)
    expect(result).toEqual({ latitude: 40.7128, longitude: -74.006 })
  })

  it('rejects latitude out of range (> 90)', () => {
    expect(validateGPSCoordinate(91, -74)).toBeNull()
  })

  it('rejects latitude out of range (< -90)', () => {
    expect(validateGPSCoordinate(-91, -74)).toBeNull()
  })

  it('rejects longitude out of range (> 180)', () => {
    expect(validateGPSCoordinate(40, 181)).toBeNull()
  })

  it('rejects longitude out of range (< -180)', () => {
    expect(validateGPSCoordinate(40, -181)).toBeNull()
  })

  it('rounds excessive precision to 6 decimals', () => {
    const result = validateGPSCoordinate(40.123456789, -74.987654321)
    expect(result).toEqual({ latitude: 40.123457, longitude: -74.987654 })
  })

  it('validates edge case: origin (0, 0)', () => {
    const result = validateGPSCoordinate(0, 0)
    expect(result).toEqual({ latitude: 0, longitude: 0 })
  })

  it('validates edge case: min values (-90, -180)', () => {
    const result = validateGPSCoordinate(-90, -180)
    expect(result).toEqual({ latitude: -90, longitude: -180 })
  })

  it('validates edge case: max values (90, 180)', () => {
    const result = validateGPSCoordinate(90, 180)
    expect(result).toEqual({ latitude: 90, longitude: 180 })
  })
})

describe('GPS_REGEX', () => {
  it('matches valid integer coordinates', () => {
    expect(GPS_REGEX.test('40')).toBe(true)
    expect(GPS_REGEX.test('-74')).toBe(true)
  })

  it('matches valid decimal coordinates', () => {
    expect(GPS_REGEX.test('40.7128')).toBe(true)
    expect(GPS_REGEX.test('-74.0060')).toBe(true)
  })

  it('matches negative zero', () => {
    expect(GPS_REGEX.test('-0')).toBe(true)
  })

  it('rejects non-numeric strings', () => {
    expect(GPS_REGEX.test('abc')).toBe(false)
    expect(GPS_REGEX.test('40.abc')).toBe(false)
  })

  it('rejects multiple decimal points', () => {
    expect(GPS_REGEX.test('40.71.28')).toBe(false)
  })

  it('allows up to 6 decimal places', () => {
    expect(GPS_REGEX.test('40.123456')).toBe(true)
  })
})

describe('parseGPSString', () => {
  it('parses valid coordinate strings', () => {
    const result = parseGPSString('40.7128', '-74.0060')
    expect(result).toEqual({ latitude: 40.7128, longitude: -74.006 })
  })

  it('rejects invalid latitude format', () => {
    expect(parseGPSString('abc', '-74.0060')).toBeNull()
  })

  it('rejects invalid longitude format', () => {
    expect(parseGPSString('40.7128', 'xyz')).toBeNull()
  })

  it('rejects out-of-range coordinates', () => {
    expect(parseGPSString('91', '-74')).toBeNull()
    expect(parseGPSString('40', '181')).toBeNull()
  })

  it('handles string integers', () => {
    const result = parseGPSString('40', '-74')
    expect(result).toEqual({ latitude: 40, longitude: -74 })
  })
})
```

Why 6 decimals (not 10):
- Commercial GPS receivers accurate to ~1.1 meters (5 decimals)
- 6 decimals = 0.11m precision (exceeds GPS accuracy slightly)
- 7+ decimals = false precision (GPS hardware can't measure <10mm)
- Database storage: 6 decimals balances accuracy and efficiency

Per 01-RESEARCH.md Pitfall 4: DO NOT store 10+ decimal GPS coordinates (wastes space, exceeds GPS device capability)

// app/components/SplitLayout/SplitLayout.tsx
import { ReactNode } from 'react'
import styles from './SplitLayout.module.css'

export interface SplitLayoutProps {
  map: ReactNode
  content: ReactNode
}

/**
 * Responsive split-screen layout for journal interface
 * 
 * Desktop (>768px): 50% map left, 50% content right (side-by-side)
 * Mobile (≤768px): Stacked with sticky map top, scrolling content below
 */
export default function SplitLayout({ map, content }: SplitLayoutProps) {
  return (
    <div className={styles.container}>
      <aside className={styles.mapPane} aria-label="Map view">
        {map}
      </aside>
      <main className={styles.contentPane}>
        {content}
      </main>
    </div>
  )
}
```

  <action>
Create lib/payload.ts with singleton pattern to prevent multiple database connections:

```typescript
// lib/payload.ts
import { getPayload as getPayloadInstance } from 'payload'
import config from '@/payload.config'

/**
 * Payload CMS client singleton
 * Prevents multiple database connections in development hot-reload
 */
let cachedPayload: any = null

export async function getPayloadClient() {
  if (cachedPayload) {
    return cachedPayload
  }

  cachedPayload = await getPayloadInstance({ config })
  return cachedPayload
}
```

Why singleton pattern:
- Next.js development hot-reload re-imports modules
- Without singleton, each hot-reload creates new database connection
- MongoDB has connection pool limits
- Singleton reuses connection across hot-reloads

Per 01-RESEARCH.md: Use singleton pattern to prevent re-mounting issues (STATE.md blocker).

```typescript
// app/page.tsx
import { getPayloadClient } from '@/lib/payload'
import SplitLayout from './components/SplitLayout/SplitLayout'
import type { DailyEntry } from '@/payload-types'

export default async function HomePage() {
  const payload = await getPayloadClient()
  
  const { docs: entries } = await payload.find({
    collection: 'daily-entries',
    sort: '-date',
    limit: 10,
  })

  const mapPlaceholder = (
    <div style={{ 
      height: '100%', 
      display: 'flex', 
      alignItems: 'center', 
      justifyContent: 'center',
      background: 'var(--journal-paper-dark)',
      color: 'var(--ink-faded)',
      fontFamily: 'var(--font-body)',
      padding: 'var(--space-xl)',
      textAlign: 'center'
    }}>
      <div>
        <h2 style={{ fontFamily: 'var(--font-display)', fontSize: '2rem', marginBottom: '1rem' }}>
          Map View
        </h2>
        <p>Map integration coming in Phase 3</p>
      </div>
    </div>
  )

  const journalContent = (
    <div>
      <h1 style={{ 
        fontFamily: 'var(--font-display)', 
        fontSize: '3.5rem',
        color: 'var(--leather-dark)',
        marginBottom: 'var(--space-2xl)'
      }}>
        Reddy on the Road
      </h1>
      
      {entries.length === 0 ? (
        <p style={{ 
          fontFamily: 'var(--font-body)',
          fontSize: '1.125rem',
          color: 'var(--ink-faded)',
          lineHeight: 1.8
        }}>
          No entries yet. Add daily entries via Payload CMS admin panel at{' '}
          <a href="/admin" style={{ color: 'var(--ink-blue)' }}>/admin</a>
        </p>
      ) : (
        <div>
          {entries.map((entry: DailyEntry) => (
            <article 
              key={entry.id}
              style={{
                background: 'var(--journal-paper)',
                border: '1px solid var(--sepia-medium)',
                borderRadius: '4px',
                padding: 'var(--space-xl)',
                marginBottom: 'var(--space-2xl)',
                boxShadow: 'var(--shadow-md)'
              }}
            >
              <h2 style={{ 
                fontFamily: 'var(--font-display)',
                fontSize: '2.5rem',
                color: 'var(--leather-dark)',
                marginBottom: 'var(--space-md)'
              }}>
                {entry.location}
              </h2>
              
              <time style={{
                fontFamily: 'var(--font-mono)',
                fontSize: '0.875rem',
                color: 'var(--ink-sepia)',
                display: 'block',
                marginBottom: 'var(--space-lg)'
              }}>
                {new Date(entry.date).toLocaleDateString('en-US', {
                  weekday: 'long',
                  year: 'numeric',
                  month: 'long',
                  day: 'numeric'
                })}
              </time>
              
              <div style={{
                fontFamily: 'var(--font-body)',
                fontSize: '1.125rem',
                lineHeight: 1.8,
                color: 'var(--ink-black)'
              }}>
                {entry.weather && (
                  <p style={{ marginBottom: 'var(--space-md)' }}>
                    <strong>Weather:</strong> {entry.weather}
                  </p>
                )}
                {entry.distanceMiles !== null && entry.distanceMiles !== undefined && (
                  <p style={{ marginBottom: 'var(--space-md)' }}>
                    <strong>Distance:</strong> {entry.distanceMiles} miles
                  </p>
                )}
              </div>
            </article>
          ))}
        </div>
      )}
    </div>
  )

  return <SplitLayout map={mapPlaceholder} content={journalContent} />
}
```

Why inline styles (not CSS Modules here):
- This is demo page showing CMS integration works
- Later phases will extract proper components with CSS Modules
- Inline styles use design tokens to prove token system works

Why async Server Component:
- Fetches data at build/request time (SSR/SSG)
- No client-side fetch needed
- SEO-friendly (HTML includes content)
- No loading spinners for initial render

Per 01-RESEARCH.md Pattern 1: Async Server Components fetch data directly from Payload CMS Local API.
